cmake_minimum_required(VERSION 3.20)
project(wavenet VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(WAVENET_SANITIZE "Build with ASan+UBSan" OFF)
option(WAVENET_FUZZ "Build libFuzzer targets (clang only)" OFF)

find_package(OpenSSL 3.0 REQUIRED)
find_package(SQLite3 REQUIRED)
find_package(Threads REQUIRED)

add_compile_options(-Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wno-sign-conversion)
if(WAVENET_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()

file(GLOB WAVENET_CORE_SRC CONFIGURE_DEPENDS core/src/*.cpp)
add_library(wavenet_core STATIC ${WAVENET_CORE_SRC})
target_include_directories(wavenet_core PUBLIC core/include)
target_link_libraries(wavenet_core PUBLIC OpenSSL::SSL OpenSSL::Crypto SQLite::SQLite3 Threads::Threads)

enable_testing()
file(GLOB WAVENET_TEST_SRC CONFIGURE_DEPENDS tests/unit/*.cpp)
add_executable(wavenet_tests ${WAVENET_TEST_SRC})
target_link_libraries(wavenet_tests PRIVATE wavenet_core)
add_test(NAME unit COMMAND wavenet_tests)

if(EXISTS ${CMAKE_SOURCE_DIR}/linux-node/CMakeLists.txt)
  add_subdirectory(linux-node)
endif()
if(WAVENET_FUZZ)
  add_subdirectory(tests/fuzz)
endif()
